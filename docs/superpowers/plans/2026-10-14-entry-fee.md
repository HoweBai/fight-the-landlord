```ts
/**
 * 台费费率表（本片第 42 键；规格 §2.1a E1–E4 逐字）：**空数组合法**；每项须对象、
 * `baseScore` 正整数且**表内唯一**、`percent` **整数** `0 ≤ n ≤ 50`（与抽水同族——整数域）。
 * **不要求覆盖档位表**（跨键关系写入口校验不了 ⇒ 未命中 ⇒ 0，fail-safe；一致性由运营备注承担）。
 */
function parseEntryFeePercents(value: unknown): EntryFeePercent[] {
  if (!Array.isArray(value)) {
    throw new ProtocolError(`currency.entryFeePercents 必须是数组（收到 ${JSON.stringify(value)}）`);
  }
  const table: EntryFeePercent[] = [];
  for (let i = 0; i < value.length; i++) {
    const item: unknown = value[i];
    if (!isObj(item) || !('baseScore' in item) || !('percent' in item)) {
      throw new ProtocolError(`currency.entryFeePercents[${i}] 必须是含 baseScore 与 percent 的对象`);
    }
    const { baseScore, percent } = item;
    if (typeof baseScore !== 'number' || !Number.isInteger(baseScore) || baseScore <= 0) {
      throw new ProtocolError(`baseScore 必须是正整数（收到 ${JSON.stringify(baseScore)}）`);
    }
    if (table.some((t) => t.baseScore === baseScore)) {
      throw new ProtocolError(`baseScore 不得重复（重复：${baseScore}）`);
    }
    if (typeof percent !== 'number' || !Number.isInteger(percent) || percent < 0 || percent > 50) {
      throw new ProtocolError(`percent 必须是 0 ≤ n ≤ 50 的整数（收到 ${JSON.stringify(percent)}）`);
    }
    table.push({ baseScore, percent });
  }
  return table;
}
```

```ts
/**
 * 台费金额（本片；客户端=服务端**同一函数**）：按 `baseScore` 精确匹配 ⇒ 未命中 ⇒ 0；
 * 命中 ⇒ `percent > 0 ? max(1, floor(baseScore * percent / 100)) : 0`。
 * 算术界：`baseScore ≤ 8191`、`percent ≤ 50` ⇒ 单笔 ≤ 4095（双精度内精确；远低于
 * `MAX_ABS_DELTA` 默认 1e6 ⇒ **无需** source 化上限）。**percent = 0 ⇒ 0 且无保底**
 * （暗上线不产生任何 1 枚误扣）。显式 `Math.floor`（照结算同族）。
 */
export function entryFeeOf(baseScore: number, percents: ReadonlyArray<EntryFeePercent>): number {
  const hit = percents.find((p) => p.baseScore === baseScore);
  if (hit === undefined || hit.percent <= 0) return 0;
  return Math.max(1, Math.floor((baseScore * hit.percent) / 100));
}
```

```ts
   * 台费收费块（本片规格 §2.3 逐字；D1–D5）。
   *
   * - **fee 快照**：fee/entries 在本次调用开头算一次，**不随重试重读配置**（14002 收窄为
   *   「跨调用并发 × 中途改费率」——r2-F1）；**该房 fee==0 整块短路**（D5：表缺失/形状坏/

   * - 单人结局四分支与免单闭环在 `chargeOne`；本方法只做「一次快照 + 并发扇出」+ 防御性外壳。

  private async chargeEntryFees(code: string, baseScore: number, seats: Seat[]): Promise<void> {

      if (fee === 0) return; // D5：该房 fee==0 ⇒ 整块跳过（不读不写不计）

      const humans = humanSeats(seats);
      await Promise.all(humans.map(({ userId, seat }) => this.chargeOne(code, baseScore, seat, userId, fee)));

  /**
   * 单人台费（规格 §2.3 逐字）：**首发 + ≤2 次重试**（退避 300ms/1s；单次 3s 超时在
   * `CurrencyClient` 内部）、**同锚同载荷**（actionId/entries 一次算定）。本函数**绝不抛**。
   *
   * 结局四分支：
   *  - `ok:true`（含 `replayed:true`）⇒ 成功；余额回写（派生键 + EXPIRE；失败只 warn）；
   *  - `14001` ⇒ 免单 `insufficient_funds`（单人 op ⇒ 归因精确到当事人）；
   *  - `14002` ⇒ **视为已收**：响亮日志 + `entry_fee_conflict_total`；**不记 skips、不计漏收**；
   *  - `10007`/超时族（重试耗尽仍无结局）⇒ 免单 `unknown_outcome`；
   *  - 其它业务码（含 10001）与其它异常 ⇒ 免单 `internal_error`（确定性失败，不重试）。
   */
  private async chargeOne(
    code: string,
    baseScore: number,
    seat: number,
    ownerId: string,
    fee: number,
  ): Promise<void> {
    const actionId = `entryfee:${code}:${ownerId}`;
    const entries = [{ ownerId, delta: -fee }];
    let lastError: unknown = null;

    for (let attempt = 0; attempt <= ENTRY_FEE_RETRY_BACKOFF_MS.length; attempt += 1) {
      if (attempt > 0) {
        await new Promise((resolve) => setTimeout(resolve, ENTRY_FEE_RETRY_BACKOFF_MS[attempt - 1]!));
      }
      try {
        const data = await this.currency.apply({ actionId, source: ENTRY_FEE_SOURCE, entries, refId: code });
        await this.writeBackFeeBalance(code, seat, ownerId, data);
        return;
      } catch (err) {
        if (err instanceof CurrencyServiceError) {
          if (err.code === ErrCode.E_INSUFFICIENT_FUNDS) {
            await this.recordEntryFeeSkip(code, seat, ownerId, baseScore, fee, 'insufficient_funds', actionId);
            return;
          }
          if (err.code === ErrCode.E_ACTION_CONFLICT) {
            // 14002 ⟺ 同锚已有**成功提交** ⇒ 该笔台费已收（「视为已收」——r2-F1/r2-delta-F2 清残留：
            // 记成漏收会让用户⑥ 的统计面系统性假漏收）。响亮日志 + 专用计数；不记 skips、不回写。
            this.logger.error(
              `entry fee conflict（同锚已有成功提交，视为已收——不记漏收）room=${code} owner=${ownerId} actionId=${actionId}`,
            );
            entryFeeConflictTotal.inc();
            return;
          }
          if (err.code === ErrCode.UPSTREAM_UNAVAILABLE) {
            lastError = err; // 10007/超时族：无结局 ⇒ 同锚同载荷重发（重放判定是唯一结局判定手段）
            continue;
          }
        }
        // 其它业务码（如 10001）与其它异常：确定性的失败 ⇒ 免单 internal_error（不重试）。
        await this.recordEntryFeeSkip(code, seat, ownerId, baseScore, fee, 'internal_error', actionId);
        return;
      }
    }
    // 首发 + ≤2 重试耗尽仍无结局 ⇒ 免单 unknown_outcome。
    this.logger.warn(`entry fee unresolved after retries room=${code} owner=${ownerId}: ${String(lastError)}`);
    await this.recordEntryFeeSkip(code, seat, ownerId, baseScore, fee, 'unknown_outcome', actionId);
  }

  /**
   * 余额回写（D2/r2-F2）：成功/重放后，以响应 `entries[].balanceAfter` HSET
   * `room:{code}:balances`（seat→值）并补 `EXPIRE ROOM_TTL_SECONDS`（照 `storeGoldBalances` 纪律）。
   * 失败 ⇒ warn（显示回落扣前值、至结算自愈——设计 §6-9；**钱面无影响**）。绝不抛。
   */
  private async writeBackFeeBalance(
    code: string,
    seat: number,
    ownerId: string,
    data: CurrencyApplyData,
  ): Promise<void> {
    try {
      const entry = data.entries.find((e) => e.ownerId === ownerId);
      if (entry === undefined) {
        this.logger.warn(`entry fee write-back skipped for ${code}: 响应缺 ownerId=${ownerId}（不写假值）`);
        return;
      }
      await this.redis.hset(roomBalancesKey(code), { [String(seat)]: String(entry.balanceAfter) });
      await this.redis.expire(roomBalancesKey(code), ROOM_TTL_SECONDS);
    } catch (err) {
      this.logger.warn(
        `entry fee write-back failed for ${code} seat=${seat}（显示回落扣前值；钱面无影响）: ${String(err)}`,
      );
    }
  }

  /**
   * 免单记账（规格 §2.4 三面 + 判别三步 r2-delta-F1；本函数绝不抛）。
   *
   * ① 逐笔日志（NDJSON 九字段 + traceId 全链）：`event='entry_fee.skipped'` +
   *    `roomCode/ownerId/seat/baseScore/levelName(反查失败省略)/feeAmount/reason/opActionId`，
   *    每笔一行、绝不合并；② 指标 `entry_fee_skipped_total{reason}`（有界 ×3）；
   * ③ 库面 skip RPC（单次尝试、不重试——防调用内重发重复；跨调用重复由唯一键 + INSERT IGNORE 兜）。
   *
   * 判别三步：`inserted=true` ⇒ inc 指标 + 逐笔日志；`false`（同房二开重复）⇒ 仅 debug、零指标；
   * **RPC 失败（无法判别）⇒ 照计指标 + warn 注明「落行失败」**（降级面：宁可多记一笔可解释读数，不漏记）。
   */
  private async recordEntryFeeSkip(
    code: string,
    seat: number,
    ownerId: string,
    baseScore: number,
    fee: number,
    reason: EntryFeeSkipReason,
    opActionId: string,
  ): Promise<void> {
    const levelName = await this.levelNameFor(baseScore);
    const fields =
      `roomCode=${code} ownerId=${ownerId} seat=${seat} baseScore=${baseScore}` +
      `${levelName === null ? '' : ` levelName=${levelName}`} feeAmount=${fee} reason=${reason} opActionId=${opActionId}`;
    try {
      const { inserted } = await this.currency.entryFeeSkip({
        roomCode: code,
        ownerId,
        seat,
        baseScore,
        feeAmount: fee,
        reason,
      });
      if (!inserted) {
        this.logger.debug(`entry_fee.skipped（同房重复，未新增行）${fields}`);
        return;
      }
      entryFeeSkippedTotal.inc({ reason });
      this.logger.warn(`entry_fee.skipped ${fields}`);
    } catch (err) {
      entryFeeSkippedTotal.inc({ reason });
      this.logger.warn(`entry_fee.skipped（落行失败）${fields} —— ${String(err)}`);
    }
  }

  /** 日志面用：baseScore → 档位名（反查失败/表不可用 ⇒ null ⇒ 字段省略——绝不为日志阻断收费链）。 */
  private async levelNameFor(baseScore: number): Promise<string | null> {
    try {
      const levels = await this.configCache.baseScoreLevels();
      return levels.find((l) => l.baseScore === baseScore)?.name ?? null;
    } catch {
      return null;
    }
  }
```

```ts
   * 本片 D6：批量读 **in-flight 声明**（`EXISTS match:claim:{playerId}`；pipeline 一次往返）。
   * 旗读纪律照 `readBannedFlags`（R11/F8）：单独 try/catch ⇒ **按无 claim** + error 日志
   * ——读失败绝不影响匹配（fail-open）；逐条错误同样按无 claim。
   */
  private async readClaimFlags(playerIds: readonly string[]): Promise<Set<string>> {
    const claimed = new Set<string>();
    if (playerIds.length === 0) return claimed;
    try {
      const pipeline = this.redis.pipeline();
      for (const id of playerIds) pipeline.exists(matchClaimKey(id));
      const results = await pipeline.exec();
      (results ?? []).forEach(([err, res], index) => {
        if (!err && res === 1) claimed.add(playerIds[index]!);
      });
      return claimed;
    } catch (err) {
      this.logger.error(`match:claim flag read failed (treating nobody as claimed): ${String(err)}`);
      return new Set();
    }
  }
```

```js
/** 台费费率表：{baseScore, percent} 行增删；空表合法（= 全部不收费）。 */
function buildEntryFeeEditor(box, ctx) {
  const rowsWrap = el('div', 'struct-rows');
  const refs = [];
  function addRow(baseScore, percent) {
    const row = el('div', 'struct-row');
    const scoreEl = document.createElement('input');
    scoreEl.type = 'number';
    scoreEl.className = 'field-num';
    scoreEl.min = '1';
    scoreEl.max = '8191';
    scoreEl.value = baseScore === undefined || baseScore === null ? '' : String(baseScore);
    const pctEl = document.createElement('input');
    pctEl.type = 'number';
    pctEl.className = 'field-num';
    pctEl.min = '0';
    pctEl.max = '50';
    pctEl.value = percent === undefined || percent === null ? '' : String(percent);
    const del = btn('删除', 'struct-remove');
    const ref = { scoreEl, pctEl, row };
    const onChange = () => {
      ctx.dirty();
      ctx.state('', '');
    };
    scoreEl.addEventListener('input', onChange);
    pctEl.addEventListener('input', onChange);
    del.addEventListener('click', () => {
      row.remove();
      const i = refs.indexOf(ref);
      if (i >= 0) refs.splice(i, 1);
      ctx.dirty();
    });
    row.append(scoreEl, el('span', 'num-suffix', '底分'), pctEl, el('span', 'num-suffix', '%'), del);
    rowsWrap.append(row);
    refs.push(ref);
  }
  for (const item of Array.isArray(ctx.snap) ? ctx.snap : []) addRow(item?.baseScore, item?.percent);
  const add = btn('+ 添加费率', 'struct-add');
  add.addEventListener('click', () => {
    addRow('', '');
    ctx.dirty();
  });
  box.append(
    rowsWrap,
    add,
    el('p', 'struct-note', '底分为正整数、不重复；费率为 0–50 的整数百分比；空表合法（= 全部不收费）。'),
  );
  ctx.bind(
    () => {
      const out = [];
      for (const r of refs) {
        const score = Number(r.scoreEl.value);
        const percentRaw = r.pctEl.value.trim();
        if (!Number.isInteger(score) || score <= 0 || score > 8191) {
          throw new Error('底分须为 1–8191 的正整数');
        }
        if (percentRaw === '' || !Number.isInteger(Number(percentRaw)) || Number(percentRaw) < 0 || Number(percentRaw) > 50) {
          throw new Error('费率须为 0–50 的整数百分比（不收该档请填 0）');
        }
        if (out.some((o) => o.baseScore === score)) throw new Error('底分不得重复');
        out.push({ baseScore: score, percent: Number(percentRaw) });
      }
      return out; // 空表合法：全部删除 ⇒ []（= 全部不收费）
    },
    () => {
      rowsWrap.textContent = '';
      refs.length = 0;
      for (const item of Array.isArray(ctx.snap) ? ctx.snap : []) addRow(item?.baseScore, item?.percent);
      ctx.state('', '');
    },
  );
}
```

```ts
/** 财务汇总的**源行**读数（共享脏栈 ⇒ 只断增量；行缺席 ⇒ 0/0）。 */
async function summarySourceDelta(
  admin: string,
  source: string,
  from: number,
  to: number,
): Promise<{ delta: number; flowCount: number }> {
  const res = await getJson<{ rows: Array<{ k: string; delta: number; flowCount: number }> }>(
    `/api/admin/wallet/summary?from=${from}&to=${to}`,
    admin,
  );
  if (res.body?.code !== ErrCode.OK) {
    throw new ConformanceError(`GET /api/admin/wallet/summary 失败：HTTP ${res.status} ${JSON.stringify(res.body)}`);
  }
  const row = (res.body.data?.rows ?? []).find((r) => r.k === source);
  return { delta: row?.delta ?? 0, flowCount: row?.flowCount ?? 0 };
}

/** 财务汇总的**漏收块**读数（同上：只断增量）。 */

/** 管理面拆笔预充（照 slice-18 runGoldGame 的形态；返回**最终**读数）。 */
async function precharge(admin: string, userId: string, label: string): Promise<number | null> {
  let lastBalance: number | null = null;
  for (let op = 1; op <= PRE_CHARGE_OPS; op += 1) {
    const res = await adminAdjust(admin, userId, PRE_CHARGE_STEP, AID(`${label}-pre-${op}`));
    if (res.status !== 200 || res.code !== ErrCode.OK) {
      throw new ConformanceError(`预充第 ${op} 笔失败 HTTP ${res.status} code=${res.code}（user=${userId}）`);
    }
    lastBalance = res.balanceAfter;
  }
  return lastBalance;
}
```

````markdown
## 一、「有局无费」——有金币局记录、但该真人既没有台费 op 也没有免单行

覆盖形态：① `game.start` 成功与逐人收费之间**进程崩**（设计 §6-1/§6-11，含「逐人之间崩 ⇒ 部分收取」）；
② 旧版本窗口开的局（当时还没有收费块）；③ 未来若代码回归导致收费块整体未执行。

```sql
SELECT r.room_code, p.user_id, r.ended_at
FROM game_record_players p
JOIN game_records r ON r.id = p.record_id
LEFT JOIN currency_ops o
  ON o.action_id = CONCAT('entryfee:', r.room_code, ':', p.user_id)
LEFT JOIN currency_entry_fee_skips s
  ON s.room_code = r.room_code AND s.owner_id = p.user_id
WHERE r.mode = 'gold' AND p.is_robot = 0
  AND o.id IS NULL AND s.id IS NULL
ORDER BY r.ended_at DESC;
```

- 期望：**空集**。非空 ⇒ 平台白送台费（对玩家无害）；逐行看 `ended_at` 与上线时间对照分类（① 崩溃窗口 / ② 旧版本窗口 / ③ 疑似缺陷 ⇒ 上报）。
````
