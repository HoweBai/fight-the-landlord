# 注册必填昵称 + 牌桌对手昵称 · 实施计划 v1.0

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**目标：** 注册时必填昵称，并让牌桌（含好友房）显示同桌玩家的昵称。

**架构：** 昵称唯一真源 = `users.nickname`；room-service 经新增的 `user.nicknames.get` TCP RPC 批量读取并缓存到 Redis `room:{code}:nicknames`；`RoomPlayer.nickname?` 随 `room.join` ack 与补发的 `room.state_update` 下发；客户端按座位渲染，缺省回落「玩家N」。

**技术栈：** NestJS 11 · TypeORM · ioredis · vitest · Vue 3 + Pinia · uni-app。

```ts
import { Controller, Logger } from '@nestjs/common';
import { MessagePattern, Payload } from '@nestjs/microservices';
import {
  ErrCode,
  RPC_PATTERNS,
  type UserNicknamesGetRequest,
  type UserNicknamesGetResponse,
} from '@fight-the-landlord/protocol';
import { UsersRepository } from '../users/users.repository';

/** 单次批量读上限（一间房最多 3 个；留余量给未来复用）。 */
export const USER_NICKNAMES_MAX_BATCH = 50;

/**
 * `user.nicknames.get` 接收侧（昵称片）：批量读昵称，**只读、无写边**。
 *
 * - 非法载荷 / 超上限 ⇒ `{ok:false, BAD_PARAM}`；
 * - 空数组 ⇒ 不查库，直接返回空列表；
 * - 不存在的 userId ⇒ 该条 `nickname: null`（不是漏项）；
 * - 数据库故障 ⇒ `{ok:false, UPSTREAM_UNAVAILABLE}`（与 `record.write` 同口径）。
 */
@Controller()
export class UserNicknamesRpcController {
  private readonly logger = new Logger(UserNicknamesRpcController.name);

  constructor(private readonly usersRepo: UsersRepository) {}

  @MessagePattern(RPC_PATTERNS.userNicknamesGet)
  async get(@Payload() dto: UserNicknamesGetRequest): Promise<UserNicknamesGetResponse> {
    if (typeof dto !== 'object' || dto === null || !Array.isArray(dto.userIds)) {
      return { ok: false, code: ErrCode.BAD_PARAM, message: 'userIds 必须是数组' };
    }
    if (dto.userIds.length > USER_NICKNAMES_MAX_BATCH) {
      return { ok: false, code: ErrCode.BAD_PARAM, message: `userIds 最多 ${USER_NICKNAMES_MAX_BATCH} 个` };
    }
    if (dto.userIds.some((id) => typeof id !== 'string' || id.length === 0)) {
      return { ok: false, code: ErrCode.BAD_PARAM, message: 'userIds 的元素必须是非空字符串' };
    }
    if (dto.userIds.length === 0) return { ok: true, data: { nicknames: [] } };

    try {
      const rows = await this.usersRepo.findNicknamesByIds(dto.userIds);
      const byId = new Map(rows.map((row) => [row.id, row.nickname]));
      return {
        ok: true,
        data: {
          nicknames: dto.userIds.map((userId) => ({ userId, nickname: byId.get(userId) ?? null })),
        },
      };
    } catch (err) {
      const message = err instanceof Error ? err.message : String(err);
      this.logger.error(`user.nicknames.get 失败：${message}`);
      return { ok: false, code: ErrCode.UPSTREAM_UNAVAILABLE, message };
    }
  }
}
```

```ts
import { Injectable, Logger } from '@nestjs/common';
import { ClientProxy } from '@nestjs/microservices';
import {
  RPC_PATTERNS,
  type UserNicknamesGetResponse,
} from '@fight-the-landlord/protocol';
import { metaFromTrace } from '@fight-the-landlord/shared';
import { lastValueFrom, timeout } from 'rxjs';

/** 昵称片：与 `record.write` 同一条边、同一预算（3s / 不重试）。 */
export const NICKNAME_TIMEOUT_MS = 3000;

/**
 * room-service → user-service 的昵称读客户端（`user.nicknames.get`）。
 *
 * ⚠️ **失败不抛**：昵称是纯展示字段 —— RPC 故障只让牌桌回落「玩家N」，
 * 绝不能拖垮入房/开局。原因只进 warn 日志。
 */
@Injectable()
export class NicknameClient {
  private readonly logger = new Logger(NicknameClient.name);

  constructor(private readonly proxy: ClientProxy) {}

  async getNicknames(userIds: string[]): Promise<Map<string, string | null>> {
    const result = new Map<string, string | null>();
    if (userIds.length === 0) return result;

    try {
      const res = await lastValueFrom(
        this.proxy
          // meta-on-every-rpc-send 闸：发送点必须带 metaFromTrace()
          .send<UserNicknamesGetResponse>(RPC_PATTERNS.userNicknamesGet, {
            userIds,
            ...metaFromTrace(),
          })
          .pipe(timeout(NICKNAME_TIMEOUT_MS)),
      );
      if (!res?.ok) {
        this.logger.warn(
          `user-service RPC ${RPC_PATTERNS.userNicknamesGet} failed: code=${String(res?.code)} ${res?.message ?? ''}`,
        );
        return result;
      }
      for (const entry of res.data.nicknames) result.set(entry.userId, entry.nickname);
      return result;
    } catch (err) {
      const reason = err instanceof Error ? err.message : String(err);
      this.logger.warn(`user-service RPC ${RPC_PATTERNS.userNicknamesGet} failed: ${reason}`);
      return result;
    }
  }
}
```

```ts
import { Inject, Injectable } from '@nestjs/common';
import type { Seat } from '@fight-the-landlord/protocol';
import Redis from 'ioredis';
import { NicknameClient } from '../gateway/nickname.client';
import { nicknamesKey, ROOM_TTL_SECONDS } from './rooms.seats';

/**
 * 房间昵称缓存（昵称片 · 设计 D4）。
 *
 * `ensure` 是唯一入口：读 `room:{code}:nicknames`，只对**缺失的人类 userId** 批量回源；
 * 成功后把 `nickname ?? ''` 写回（空串 = 已知无昵称）。回源失败（空 map）**不写缓存**，
 * 下次状态推送再试。返回的 Map **只含非空昵称**（调用方据此决定是否加字段）。
 */
@Injectable()
export class RoomNicknamesService {
  constructor(
    @Inject(Redis) private readonly redis: Redis,
    private readonly client: NicknameClient,
  ) {}

  async ensure(code: string, players: readonly Seat[]): Promise<Map<string, string>> {
    const key = nicknamesKey(code);
    const cached = new Map<string, string | null>();
    for (const [userId, value] of Object.entries((await this.redis.hgetall(key)) ?? {})) {
      cached.set(userId, value === '' ? null : value);
    }

    const humanIds = [...new Set(players.filter((p) => !p.isRobot).map((p) => p.userId))];
    const missing = humanIds.filter((userId) => !cached.has(userId));

    if (missing.length > 0) {
      const fetched = await this.client.getNicknames(missing);
      const fields: Record<string, string> = {};
      for (const userId of missing) {
        if (!fetched.has(userId)) continue;
        const nickname = fetched.get(userId) ?? null;
        cached.set(userId, nickname);
        fields[userId] = nickname ?? '';
      }
      if (Object.keys(fields).length > 0) {
        await this.redis.hset(key, fields);
        await this.redis.expire(key, ROOM_TTL_SECONDS);
      }
    }

    const out = new Map<string, string>();
    for (const [userId, nickname] of cached) if (nickname) out.set(userId, nickname);
    return out;
  }
}
```
